cmake_minimum_required(VERSION 3.22)
project(thor_rife LANGUAGES C CXX)
add_library(thor_mpv SHARED IMPORTED)
set_target_properties(thor_mpv PROPERTIES IMPORTED_LOCATION "${THOR_MPV_DIR}/libmpv.so")
add_library(thor_avcodec SHARED IMPORTED)
set_target_properties(thor_avcodec PROPERTIES IMPORTED_LOCATION "${THOR_MPV_DIR}/libavcodec.so")
add_library(thor_dovi SHARED dovi_player.cpp)
target_compile_features(thor_dovi PRIVATE cxx_std_17)
target_compile_options(thor_dovi PRIVATE -fexceptions -Wall -Wextra)
target_link_options(thor_dovi PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,defs)
target_link_libraries(thor_dovi PRIVATE thor_mpv thor_avcodec android nativewindow log)
add_library(thor_subtitle_audio SHARED subtitles/audio_jni.cpp)
target_compile_features(thor_subtitle_audio PRIVATE cxx_std_17)
target_compile_options(thor_subtitle_audio PRIVATE -fexceptions -Wall -Wextra)
target_link_options(thor_subtitle_audio PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,defs)
target_link_libraries(thor_subtitle_audio PRIVATE thor_mpv)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Android-21 prebuilts omit AHardwareBuffer entry points. Build the pinned source
# at API 30, with the inference operators used by this model and GPU converters.
set(NCNN_VULKAN ON CACHE BOOL "" FORCE)
# Inference runs on Vulkan/QNN and CPU packing is single-threaded. Android's
# libomp affinity initialization can abort while loading a model on a new worker;
# omit that unused runtime instead of relying on process-wide KMP environment.
set(NCNN_OPENMP OFF CACHE BOOL "" FORCE)
set(NCNN_SIMPLEVK ON CACHE BOOL "" FORCE)
set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NCNN_INT8 OFF CACHE BOOL "" FORCE)
set(NCNN_BF16 OFF CACHE BOOL "" FORCE)
set(NCNN_ARM82 OFF CACHE BOOL "" FORCE)
set(NCNN_RUNTIME_CPU OFF CACHE BOOL "" FORCE)
file(STRINGS "${THOR_NCNN_SOURCE}/src/CMakeLists.txt" LAYER_LINES REGEX "^ncnn_add_layer\\(")
set(RIFE_LAYERS binaryop concat convolution crop deconvolution eltwise input interp memorydata pixelshuffle relu sigmoid split packing cast padding)
foreach(LINE ${LAYER_LINES})
 string(REGEX REPLACE "^ncnn_add_layer\\(([A-Za-z0-9_]+).*" "\\1" LAYER "${LINE}")
 string(TOLOWER "${LAYER}" LAYER)
 if(LAYER IN_LIST RIFE_LAYERS)
  set(WITH_LAYER_${LAYER} ON CACHE BOOL "" FORCE)
 else()
  set(WITH_LAYER_${LAYER} OFF CACHE BOOL "" FORCE)
 endif()
endforeach()
add_subdirectory("${THOR_NCNN_SOURCE}" ncnn)
# Keep exceptions at our JNI boundary; ncnn itself remains exception-free.
get_target_property(NCNN_FLAGS ncnn INTERFACE_COMPILE_OPTIONS)
list(REMOVE_ITEM NCNN_FLAGS "-fno-exceptions")
set_target_properties(ncnn PROPERTIES INTERFACE_COMPILE_OPTIONS "${NCNN_FLAGS}")
# SimpleVK omits this Android foreign-queue constant; match the NDK Vulkan headers.
target_compile_definitions(ncnn PRIVATE "VK_QUEUE_FAMILY_FOREIGN_EXT=(~2U)")
# Keep ncnn exception-free without exporting that restriction to our JNI error boundary.
get_target_property(NCNN_FLAGS ncnn INTERFACE_COMPILE_OPTIONS)
if(NCNN_FLAGS)
 list(REMOVE_ITEM NCNN_FLAGS "-fno-exceptions")
 set_target_properties(ncnn PROPERTIES INTERFACE_COMPILE_OPTIONS "${NCNN_FLAGS}")
endif()
set(RIFE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/rife")
set(SHADER_HEADERS)
foreach(NAME warp warp_pack4 warp_pack8 rife_preprocess rife_postprocess rife_compose rife_preprocess_qnn rife_texture_import rife_texture_prepare rife_texture_prepare_ncnn rife_texture_flow_separate rife_texture_flow rife_texture_flow_fused rife_texture_flow_full rife_texture_compose rife_compose_qnn rife_hdr_proxy)
 set(SHADER_DIR "${RIFE_DIR}")
 if(NAME MATCHES "^rife_")
  set(SHADER_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
 endif()
 set(HEADER "${CMAKE_CURRENT_BINARY_DIR}/${NAME}.comp.hex.h")
 add_custom_command(OUTPUT "${HEADER}"
  COMMAND "${CMAKE_COMMAND}" "-DSHADER_SRC=${SHADER_DIR}/${NAME}.comp" "-DSHADER_COMP_HEADER=${HEADER}" -P "${RIFE_DIR}/generate_shader_comp_header.cmake"
  DEPENDS "${SHADER_DIR}/${NAME}.comp" "${RIFE_DIR}/generate_shader_comp_header.cmake" VERBATIM)
 list(APPEND SHADER_HEADERS "${HEADER}")
endforeach()
add_library(thor_rife SHARED rife_jni.cpp rife_image_reader.cpp vulkan_presenter.cpp "${RIFE_DIR}/warp.cpp" ${SHADER_HEADERS})
target_include_directories(thor_rife PRIVATE "${RIFE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(thor_rife PRIVATE -fexceptions -Wall -Wextra -Wno-missing-field-initializers)
target_link_options(thor_rife PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,defs)
target_link_libraries(thor_rife PRIVATE ncnn android nativewindow log vulkan dl neuralnetworks mediandk)
target_include_directories(thor_rife PRIVATE "${THOR_ORT_DIR}/headers")
add_library(thor_ort SHARED IMPORTED)
set_target_properties(thor_ort PROPERTIES IMPORTED_LOCATION "${THOR_ORT_DIR}/jni/arm64-v8a/libonnxruntime.so")
target_link_libraries(thor_rife PRIVATE thor_ort)
